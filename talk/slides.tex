\begin{frame}{Berkeley, 1969}
\begin{itemize}
\item an integrated circuit cannot be breadboarded, and an internal node
      cannot be probed
\item a design error after tapeout costs a mask set: simulation becomes
      the instrument
\item Berkeley builds a series of simulators: BIAS, SLIC, TIME, CANCER
\end{itemize}
\vspace{3mm}
\begin{tile}{The machine}
A CDC 6400 mainframe: 256\,KB of memory by day, 384\,KB at night. A
large circuit is 50 nodes and 25 transistors. Jobs run in a nightly
batch; a crash kills every job behind it.
\end{tile}
\end{frame}

\begin{frame}{CANCER}
\begin{itemize}
\item a class project in Ron Rohrer's circuit simulation course; the
      grading rule: if Don Pederson approves of the program, everyone
      passes
\item already inside: sparse matrix solver, implicit integration,
      built-in device models, adjoint sensitivity
\item presented by Rohrer at the 1971 ISSCC
\end{itemize}
\vspace{3mm}
\begin{tile}{Computer Analysis of Nonlinear Circuits, Excluding Radiation}
Simulators written under defense contracts were required to evaluate the
radiation hardness of a circuit. The name was a statement that this
program never would, and took no defense money. It was Berkeley in the
sixties.
\end{tile}
\end{frame}

\begin{frame}{CANCER becomes SPICE}
\begin{center}
\begin{tabular}{ll}
\toprule
1971 & Nagel renames it: Simulation Program with Integrated Circuit Emphasis \\
1972 & released to the public domain, Pederson's condition \\
1973 & announced at the Midwest Symposium on Circuit Theory, Waterloo \\
1975 & SPICE2, Nagel's thesis: MNA, trapezoid with LTE, limiting, stepping \\
1983 & SPICE 2G6 (Ellis Cohen): the netlist format freezes \\
1993 & SPICE 3f5 (Tom Quarles): rewrite in C, gmin stepping \\
today & ngspice, Xyce; HSPICE, PSpice, Spectre, Eldo \\
\bottomrule
\end{tabular}
\end{center}
\vspace{2mm}
\begin{itemize}
\item public domain is why SPICE won: universities taught it, graduates
      wrote to Berkeley for a tape, every derivative kept the language
\item the MOSFET model arrived in a weekend when Dave Hodges came from
      Bell Labs: Shichman-Hodges, the level 1 of this talk
\end{itemize}
\end{frame}

  to[R, l=$1\,\mathrm{k\Omega}$] (4.4,0) node[ground]{};
\end{circuitikz}
\end{center}
\column{0.58\textwidth}
\[
\begin{pmatrix}
g & -g & 1 \\
-g & 2g & 0 \\
1 & 0 & 0
\end{pmatrix}
\begin{pmatrix} v_1 \\ v_2 \\ i \end{pmatrix}
=
\begin{pmatrix} 0 \\ 0 \\ 10 \end{pmatrix}

\end{columns}
\vspace{3mm}
\begin{itemize}
\item one branch current per voltage-defined element (V, L, E)
\item the branch row has a structural zero on the diagonal: pivoting is mandatory
\item ground is a dummy row 0: stamps never special-case it
\end{itemize}
\end{frame}

\begin{frame}{Sparse LU}
\locbar{sparse LU and stamp helpers}{121 lines}{437}{558}
\vspace{2mm}
\begin{itemize}
\item rows are sorted (column, value) vectors; elimination is a merge,
      fill-in falls out as the union
\end{itemize}
\vspace{2mm}
\begin{tile}{The first-entry invariant}
Every active row contains only columns $\ge k$: a column-$k$ entry is
always the \emph{first} element, so rows bucket by their first column
and pivot search reads one bucket. The factorization is linear on
ladders and trees.
\end{tile}
\vspace{2mm}
\begin{itemize}
\item a singular matrix names its column: {\footnotesize\texttt{singular
      matrix at node b, no conduction path?}}
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Two primitives carry every device}
\begin{rust}
fn contrib(&mut self, p: usize, n: usize, i: T,
           ctl: &[(usize, usize, T)], x: &[T]) {
    let mut ieq = i;
    for &(a, b, g) in ctl {
        self.quad(p, n, a, b, g);
        ieq = ieq - g * (x[a] - x[b]);
    }
    self.rhs(p, T::zero() - ieq);
    self.rhs(n, ieq);
}
\end{rust}
\vspace{1mm}
The contribution operator of Verilog-A, $I(p,n) \mathrel{{<}{+}} f(v)$;
the dual \texttt{vcontrib} writes the branch row of the voltage-defined
elements.
\end{frame}

\begin{frame}{The whole device set}
\begin{center}
\begin{tabular}{lll}
\toprule
device & primitive & controls \\
\midrule
R, C, I & \texttt{contrib} & self conductance \\
G (VCCS) & \texttt{contrib} & one control pair \\
diode & \texttt{contrib} & one junction \\
MOSFET, JFET & \texttt{contrib} & $g_m$, $g_{ds}$ \\
BJT & 3 $\times$ \texttt{contrib} & two junctions, transport \\
V, L, E & \texttt{vcontrib} & none, own current, control pair \\
\bottomrule
\end{tabular}
\end{center}
\vspace{3mm}
\begin{itemize}
\item nmos/pmos and source-drain swap in one evaluation: every derivative
      picks up the polarity twice, $s^2 = 1$, the stamps are sign-free
\item one Jacobian structure in one place; hand-derived stamps become
      verification artifacts
\end{itemize}
\end{frame}

\begin{frame}{Junction limiting}
\[
v_{crit} = V_T \ln \frac{V_T}{\sqrt{2}\, I_S}
\qquad\qquad
v \leftarrow v_{old} + V_T \ln\!\left(1 + \frac{v_{new} - v_{old}}{V_T}\right)
\]
\vspace{1mm}
\begin{itemize}
\item Newton diverges on the bare exponential: $5$\,V proposed means
      $e^{193}$ in the Jacobian
\item beyond the curvature maximum $v_{crit}$, steps become logarithmic
\item this is SPICE2's \texttt{pnjlim}, unchanged since 1975
\end{itemize}
\vspace{1mm}
\begin{tile}{Failure mode: limiting without the flag}
A common-emitter amplifier converges with the transistor off: the limiter
clamps the junction, no current flows, the deltas vanish, the test says
done. The rule: an iteration in which a limiter changed a voltage cannot
converge. A delta test alone cannot see active limiting.
\end{tile}
\end{frame}

\begin{frame}{Junction capacitances by desugaring}
\[
C(v) = C_{j0} \left(1 - \tfrac{v}{V_J}\right)^{-m}, \qquad
Q(v) = \frac{C_{j0} V_J}{1 - m}
       \left(1 - \left(1 - \tfrac{v}{V_J}\right)^{1 - m}\right)
\]
\vspace{1mm}
\begin{itemize}
\item \texttt{cjo}, \texttt{cje}/\texttt{cjc}, \texttt{cgs}/\texttt{cgd}
      desugar into internal capacitors in the parser
\item no assembly, state, timestep, or AC code knows they exist
\item charge-based, graded ($m = 0.5$) for junctions, linear for gate oxide
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Operating point fallbacks are data}
\begin{rust}
let gmin_path: Vec<(f64, f64)> =
    (2..=12).map(|k| (10f64.powi(-k), 1.0)).collect();
let src_path: Vec<(f64, f64)> =
    (1..=10).map(|k| (GMIN, k as f64 / 10.0)).collect();
for path in [gmin_path, src_path] {
    if path.iter().all(|&(g, fac)|
        newton(ckt, x, lim, &Mode::Dc { fac }, g, ITL_OP).is_some()) {
        return;
    }
}
\end{rust}
\vspace{1mm}
Both classic fallbacks are homotopy paths in the
$(g_{min}, \text{source scale})$ plane, walked with warm starts.
\end{frame}

\begin{frame}{Adaptive timestep}
\[
\varepsilon_{tr} \approx
\frac{h^3/12}{\;h(h+d_1)(h+d_1+d_2)/6 + h^3/12\;}\,\bigl(x_c - x_p\bigr),
\qquad h' = 0.9\, h\, r^{-1/3}
\]
\vspace{1mm}
\begin{itemize}
\item quadratic predictor through the last three points; corrector minus
      predictor measures $\dddot{x}$ (uniform steps: the classic $1/13$)
\item the predictor doubles as the Newton starting point
\item waveform corners are breakpoints; steps land on them exactly
\item the two predictor-less restart steps run damped backward Euler
\item when the cuts bottom out: \texttt{timestep too small}; SPICE2 said
      \texttt{TIMESTEP REDUCED TO ZILCH}
\end{itemize}
\end{frame}

\begin{frame}{Small-signal AC}
\[
(G + j\omega C)\,\hat{x} = \hat{b}
\]
\vspace{1mm}
\begin{itemize}
\item at a converged operating point the limiters pass voltages through:
      $G$ \emph{is the DC Jacobian the last Newton iteration assembled}
\item the AC path reuses \texttt{assemble}, maps to complex, adds
      $j\omega C$ and $-j\omega L$
\item one generic sparse LU over a three-method \texttt{Num} trait serves
      Newton and AC; no device model contains AC-specific code
\end{itemize}
\end{frame}
